Исходный код Pong для прошивки ESP32-S3 (ST7735)

// Фрагмент игрового цикла Pong на ESP32-S3 (Дисплей ST7735 128x128)
void renderPongGameLoop(bool isUp, bool isDown) {
    canvas.fillScreen(COLOR_BLACK);
    
    if (isUp && playerY > 2) playerY -= 4;
    if (isDown && playerY < 108) playerY += 4;

    ballX += ballVX; ballY += ballVY;
    if (ballY <= 2 || ballY >= 124) ballVY = -ballVY;

    // ИИ ракетка
    if (aiY + 8 < ballY) aiY += 2;
    else if (aiY + 8 > ballY) aiY -= 2;

    // Отрисовка
    canvas.fillRect(4, playerY, 3, 18, COLOR_GREEN);
    canvas.fillRect(121, aiY, 3, 18, COLOR_RED);
    canvas.fillCircle((int)ballX, (int)ballY, 2, COLOR_WHITE);
    canvasFlush();
}
