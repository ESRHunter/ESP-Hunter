Быстрый Код Инициализации SPI Bus (C++)

#include <SPI.h>

#define TFT_SCLK 12
#define TFT_MOSI 11
#define SD_MISO  13
#define TFT_CS   10
#define SD_CS    15

void setup() {
    // Единая шина SPI для дисплея и MicroSD
    SPI.begin(TFT_SCLK, SD_MISO, TFT_MOSI, -1);
    
    pinMode(TFT_CS, OUTPUT); digitalWrite(TFT_CS, HIGH);
    pinMode(SD_CS, OUTPUT);  digitalWrite(SD_CS, HIGH);
}
